        # Possible Values:
        #   - DAY

    spine_mri_date = Optional mention of [Plaintext string]
        # Spine metastatic assessment date, ISO date at supported precision.

        # Possible Values:

        # Possible Values:
        #   - POSITIVE
        #   - NEGATIVE
        #   - NONE_OF_THE_ABOVE
    brain_mri_findings = MetastasisEvidence (Choice Value)
        # Brain MRI for intracranial metastasis/seeding beyond the primary. POSITIVE: intracranial seeding present. NEGATIVE: done, none. NONE_OF_THE_ABOVE: not done, indeterminate, or not reported.

        #   - NEGATIVE
        #   - NONE_OF_THE_ABOVE
    csf_cytology = MetastasisEvidence (Choice Value)
        # CSF cytology; record lumbar versus ventricular source and collection date separately. POSITIVE: malignant cells present. NEGATIVE: cytology negative. NONE_OF_THE_ABOVE: not performed, indeterminate, or not reported.
        # Possible Values:
        #   - POSITIVE
        #   - NEGATIVE
        #   - NONE_OF_THE_ABOVE
    extraneural_metastasis = MetastasisEvidence (Choice Value)
        # Metastasis outside the CNS (bone marrow, bone, viscera). POSITIVE: extraneural metastasis present. NEGATIVE: worked up, none. NONE_OF_THE_ABOVE: no extraneural workup, indeterminate, or not reported.
        # Possible Values:
        #   - POSITIVE
        #   - NEGATIVE
        #   - NONE_OF_THE_ABOVE

    site = MetastaticSite (Choice Value)
        # CSF: positive CSF/cytology. SPINE: spinal drop metastases. BRAIN: intracranial metastasis beyond primary. LEPTOMENINGEAL: leptomeningeal spread. BONE_MARROW: marrow involvement. KIDNEY: renal metastasis. LIVER: hepatic metastasis. NONE_OF_THE_ABOVE: a documented metastatic site not listed above, or the site is not stated.
        # Possible Values:
        #   - CSF
        #   - SPINE
        #   - BRAIN
        #   - LEPTOMENINGEAL
        #   - BONE_MARROW
        #   - KIDNEY
        #   - LIVER
        #   - NONE_OF_THE_ABOVE
    site_date = Optional mention of [Plaintext string]
        # Date this metastatic site was documented, ISO YYYY-MM-DD. Null if not stated.
    site_date_precision = Optional mention of [DatePrecision (Choice Value)]
        # Precision for site_date. Null when site_date is null.
        # Possible Values:
        #   - DAY

        #   - YEAR
