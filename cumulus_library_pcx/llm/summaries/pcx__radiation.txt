model = "RadiationAnnotation"
radiation_rounds: A (possibly empty) list of RadiationRoundMention instances, which contain:
    delivery_status = DeliveryStatus (Choice Value)
        # Whether this radiation course was actually delivered. Choose exactly one. A course counts as delivered once any fraction is given; plans, recommendations and simulations are not exposure. ADMINISTERED: at least one fraction was delivered, including a course stopped early. PLANNED: recommended, consented, simulated or scheduled, with no fraction delivered yet. HELD: a scheduled course paused or deferred before the first fraction, with intent to proceed. CANCELLED: a planned course explicitly cancelled before any fraction was delivered. EXPLICITLY_NOT_RECEIVED: the note states the patient did not receive radiation, for example declined, omitted by protocol, or never given. NONE_OF_THE_ABOVE: delivery cannot be determined from this note; not evidence of non-receipt.
        # Possible Values:
        #   - ADMINISTERED
        #   - PLANNED
        #   - HELD
        #   - CANCELLED
        #   - EXPLICITLY_NOT_RECEIVED
        #   - NONE_OF_THE_ABOVE
    phase = TreatmentPhase (Choice Value)
        # Possible Values:
        #   - INDUCTION
        #   - CONSOLIDATION
        #   - POST_CONSOLIDATION
        #   - SALVAGE
        #   - NONE_OF_THE_ABOVE
    indication = Optional mention of [Plaintext string]
        # Initial treatment, post-chemotherapy residual/metastatic disease, salvage after relapse or other documented indication. Do not infer from dose.
    assessed_through_date = Optional mention of [Plaintext string]
        # Date through which explicit non-receipt or delivery is assessed.
    assessed_through_date_precision = Optional mention of [DatePrecision (Choice Value)]
        # Precision for assessed_through_date; null if absent.
        # Possible Values:
        #   - DAY
        #   - MONTH
        #   - YEAR
    dose_verbatim = Optional mention of [Plaintext string]
        # Exact delivered dose(s), field(s) and units; preserve mixed units here instead of assigning incorrect shared units.
    radiation_method = RadiationMethod (Choice Value)
        # Delivery method/energy. PHOTON: photon modality. PROTON: proton modality. COMBINATION_PROTONS_AND_PHOTONS: combination of protons and photons. ELECTRONS: electron modality. THREE_D_CONFORMAL: 3D conformal. IMRT: intensity-modulated. STEREOTACTIC_RADIOSURGERY: SRS. GAMMA_KNIFE: Gamma Knife. BRACHYTHERAPY: brachytherapy. NONE_OF_THE_ABOVE: method not established; not evidence of non-receipt; or none of the above.
        # Possible Values:
        #   - PHOTON
        #   - PROTON
        #   - COMBINATION_PROTONS_AND_PHOTONS
        #   - ELECTRONS
        #   - THREE_D_CONFORMAL
        #   - IMRT
        #   - STEREOTACTIC_RADIOSURGERY
        #   - GAMMA_KNIFE
        #   - BRACHYTHERAPY
        #   - NONE_OF_THE_ABOVE
    radiation_field = RadiationField (Choice Value)
        # Treated field. CRANIOSPINAL: CSI without a documented boost. CRANIOSPINAL_WITH_FOCAL_BOOST: craniospinal irradiation plus a tumor-bed boost. FOCAL_TUMOR_BED: focal/involved-field only. WHOLE_VENTRICULAR_WITH_FOCAL_BOOST: whole-ventricular plus boost (germ-cell). NONE_OF_THE_ABOVE: a documented field not listed above, or the field is not stated.
        # Possible Values:
        #   - CRANIOSPINAL
        #   - CRANIOSPINAL_WITH_FOCAL_BOOST
        #   - FOCAL_TUMOR_BED
        #   - WHOLE_VENTRICULAR_WITH_FOCAL_BOOST
        #   - NONE_OF_THE_ABOVE
    radiation_start_date = Optional mention of [Plaintext string]
        # Round start date, ISO YYYY-MM-DD. Null if not stated.
    radiation_start_date_precision = Optional mention of [DatePrecision (Choice Value)]
        # Precision for radiation_start_date. Null when it is null.
        # Possible Values:
        #   - DAY
        #   - MONTH
        #   - YEAR
    radiation_end_date = Optional mention of [Plaintext string]
        # Round end date, ISO YYYY-MM-DD. Null if not stated.
    radiation_end_date_precision = Optional mention of [DatePrecision (Choice Value)]
        # Precision for radiation_end_date. Null when it is null.
        # Possible Values:
        #   - DAY
        #   - MONTH
        #   - YEAR
    focal_dose_to_primary_site = Optional mention of [Decimal number 0 or greater]
        # Focal boost dose to the primary tumor site (numeric, in dose_units). Null if not stated.
    total_dose_to_primary_site = Optional mention of [Decimal number 0 or greater]
        # Total dose to the primary site (focal + craniospinal/whole-ventricular if given). Null if not stated.
    focal_dose_to_metastatic_site = Optional mention of [Decimal number 0 or greater]
        # Focal boost dose to a metastatic site (only if radiation given to metastasis). Null otherwise.
    total_dose_to_metastatic_site = Optional mention of [Decimal number 0 or greater]
        # Total dose to a metastatic site (only if radiation given to metastasis). Null otherwise.
    craniospinal_dose = Optional mention of [Decimal number 0 or greater]
        # Craniospinal-axis dose (only if craniospinal radiation given). Null otherwise.
    whole_ventricular_dose = Optional mention of [Decimal number 0 or greater]
        # Whole-ventricular dose (only for germ-cell tumors). Null otherwise.
    dose_units = Optional mention of [RadiationUnits (Choice Value)]
        # Units for the dose fields in this round. CGY: centigray. GY: gray (=100 cGy). CGE: cobalt gray equivalent (proton). Null if no dose is stated or units are mixed; preserve mixed units in dose_verbatim.
        # Possible Values:
        #   - CGY
        #   - GY
        #   - CGE
