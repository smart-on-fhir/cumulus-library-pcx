results: A (possibly empty) list of LaboratoryResultMention instances, which contain:
    value_numeric = Optional mention of [Decimal number]
        # Numeric value of the result as reported in the source. Capture the point estimate only; do not infer values from ranges or qualitative statements.
    value_type = LabValueType (Choice Value)
        # Representation of the reported value. Use ABSOLUTE when the result is reported as a numeric quantity or concentration. Use PERCENT only when the result is explicitly reported as a percentage or with the '%' symbol (e.g., neutrophils 62%, eosinophils 4%).
        # Possible Values:
        #   - ABSOLUTE
        #   - PERCENT
        #   - NONE_OF_THE_ABOVE
    value_unit = Optional mention of [Plaintext string]
        # Unit of measurement verbatim from the source (e.g., 'g/dL', 'ng/mL', 'mg/L', 'ug/g', 'fL', 'x10^9/L'). Do not normalize.
    trend = LabTrend (Choice Value)
        # Physician-documented directional trajectory of this result as stated or clearly implied in clinical narrative. Capture only explicit clinician assertions about change over time (e.g., 'hemoglobin improving', 'CRP continues to rise'). Do not infer trend from serial numeric values alone.
        # Possible Values:
        #   - IMPROVING
        #   - WORSENING
        #   - STABLE
        #   - NORMALIZING
        #   - FLUCTUATING
        #   - NONE_OF_THE_ABOVE
    interp_instrument = LabInterpretation (Choice Value)
        # Structured interpretation flag as printed on the lab report or generated by the lab instrument. Capture only when an explicit flag appears in the structured result header or result table; do not infer from the numeric value alone. Choose exactly one. LOW: flagged 'L' or 'Low'. NORMAL: flagged 'N' or 'Normal', or explicitly marked within range. HIGH: flagged 'H' or 'High'. CRITICAL_LOW: flagged 'LL', 'Critical Low' or a panic-low flag. CRITICAL_HIGH: flagged 'HH', 'Critical High' or a panic-high flag. ABNORMAL: flagged 'A' or 'Abnormal' with no direction. NOT_INTERPRETED: no flag is printed for this result.
        # Possible Values:
        #   - LOW
        #   - NORMAL
        #   - HIGH
        #   - CRITICAL_LOW
        #   - CRITICAL_HIGH
        #   - ABNORMAL
        #   - NOT_INTERPRETED
    interp_clinician = LabInterpretation (Choice Value)
        # Clinician's qualitative assessment of the result as stated in the clinical narrative. Capture only explicit clinician assertions; do not infer from the numeric value alone. Choose exactly one. LOW: stated as low, decreased or below range (e.g., 'hemoglobin is low'). NORMAL: stated as normal or within normal limits (e.g., 'potassium within normal limits'). HIGH: stated as high, elevated or above range (e.g., 'CRP markedly elevated'). CRITICAL_LOW: stated as critically, dangerously or severely low. CRITICAL_HIGH: stated as critically, dangerously or severely high. ABNORMAL: stated as abnormal with no direction. NOT_INTERPRETED: the narrative reports the value without assessing it.
        # Possible Values:
        #   - LOW
        #   - NORMAL
        #   - HIGH
        #   - CRITICAL_LOW
        #   - CRITICAL_HIGH
        #   - ABNORMAL
        #   - NOT_INTERPRETED
    interp_llm = LabInterpretation (Choice Value)
        # LLM best assertion of the result interpretation, synthesized from all available context: numeric value, unit, stated reference range, patient age, clinical narrative, and any implicit cues in the document. Apply general pediatric reference ranges when no explicit range is provided. This field may be populated even when interp_instrument and interp_clinician are absent. Choose exactly one. LOW: below the applicable reference range. NORMAL: within the applicable reference range. HIGH: above the applicable reference range. CRITICAL_LOW: below a critical low limit stated in the document or generally accepted for the analyte. CRITICAL_HIGH: above a critical high limit stated in the document or generally accepted for the analyte. ABNORMAL: abnormal but the direction cannot be determined (e.g., a qualitative result). NOT_INTERPRETED: only when insufficient context exists to make any reasonable determination.
        # Possible Values:
        #   - LOW
        #   - NORMAL
        #   - HIGH
        #   - CRITICAL_LOW
        #   - CRITICAL_HIGH
        #   - ABNORMAL
        #   - NOT_INTERPRETED
    test = LaboratoryTest (Choice Value)
        # Which study-specific test this result is. Only tests in LaboratoryTest are extracted; omit results for any other test.
        # Possible Values:
        #   - ABSOLUTE_NEUTROPHIL_COUNT
        #   - PLATELET_COUNT
        #   - HEMOGLOBIN
        #   - CREATININE_CLEARANCE
        #   - MEASURED_GFR
        #   - SERUM_CREATININE
        #   - TOTAL_BILIRUBIN
        #   - AST
        #   - ALT
        #   - METHOTREXATE_LEVEL
    reference_range = Optional mention of [Plaintext string]
        # Documented normal range or upper limit of normal, including age context.
    collection_date = Optional mention of [Plaintext string]
        # Specimen/measurement date, ISO date at supported precision.
    collection_date_precision = Optional mention of [DatePrecision (Choice Value)]
        # Precision for collection_date; null if absent.
        # Possible Values:
        #   - DAY
        #   - MONTH
        #   - YEAR
