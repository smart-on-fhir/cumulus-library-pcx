        #   - NONE_OF_THE_ABOVE

    anchor_date_precision = Optional mention of [DatePrecision (Choice Value)]

        # Possible Values:
        #   - DAY

        #   - YEAR

        # Named trial/protocol for enrollment or treatment dates. Enrollment requires explicit documentation.
