model = "SystemicTherapyAnnotation"
regimens: A (possibly empty) list of MedicalTherapyRegimenMention instances, which contain:
    phase = TreatmentPhase (Choice Value)
        # Possible Values:
        #   - INDUCTION
        #   - CONSOLIDATION
        #   - POST_CONSOLIDATION
        #   - SALVAGE
        #   - NONE_OF_THE_ABOVE
    documented_trial_arm = Optional mention of [Plaintext string]
        # Explicit randomized arm assignment, if any. Do not infer randomization from observed treatment.
    protocol_name_verbatim = Optional mention of [Plaintext string]
        # Treatment protocol name as written (e.g. 'ACNS0334', 'CCG99703', 'Head Start'). Null if the note states the patient is not treated per a protocol, or no protocol is named.
    regimen_start_date = Optional mention of [Plaintext string]
        # Regimen start date, ISO YYYY-MM-DD. Null if not stated.
    regimen_start_date_precision = Optional mention of [DatePrecision (Choice Value)]
        # Precision for regimen_start_date. Null when it is null.
        # Possible Values:
        #   - DAY
        #   - MONTH
        #   - YEAR
    regimen_stop_date = Optional mention of [Plaintext string]
        # Regimen stop date, ISO YYYY-MM-DD. Null if not stated.
    regimen_stop_date_precision = Optional mention of [DatePrecision (Choice Value)]
        # Precision for regimen_stop_date. Null when it is null.
        # Possible Values:
        #   - DAY
        #   - MONTH
        #   - YEAR
    agents: A (possibly empty) list of TherapyAgentMention instances, which contain:
        delivery_status = DeliveryStatus (Choice Value)
            # Receipt of this agent when individual doses are not detailed. Plans do not establish exposure.
            # Possible Values:
            #   - ADMINISTERED
            #   - PLANNED
            #   - HELD
            #   - CANCELLED
            #   - EXPLICITLY_NOT_RECEIVED
            #   - NONE_OF_THE_ABOVE
        agent_name = Optional mention of [Plaintext string]
            # Therapeutic agent name as written (generic preferred). Null if none documented.
        therapy_start_date = Optional mention of [Plaintext string]
            # Agent start date, ISO YYYY-MM-DD. Null if not stated.
        therapy_start_date_precision = Optional mention of [DatePrecision (Choice Value)]
            # Precision for therapy_start_date. Null when it is null.
            # Possible Values:
            #   - DAY
            #   - MONTH
            #   - YEAR
        therapy_stop_date = Optional mention of [Plaintext string]
            # Agent stop date, ISO YYYY-MM-DD. Null if not stated.
        therapy_stop_date_precision = Optional mention of [DatePrecision (Choice Value)]
            # Precision for therapy_stop_date. Null when it is null.
            # Possible Values:
            #   - DAY
            #   - MONTH
            #   - YEAR
        assessed_through_date = Optional mention of [Plaintext string]
            # Date through which explicit non-receipt or delivery of this agent is assessed, ISO YYYY-MM-DD. Mainly for EXPLICITLY_NOT_RECEIVED: the note states the agent was not given as of this date. Never infer from the note date. Null if not stated.
        assessed_through_date_precision = Optional mention of [DatePrecision (Choice Value)]
            # Precision for assessed_through_date. Null when it is null.
            # Possible Values:
            #   - DAY
            #   - MONTH
            #   - YEAR
        administrations: A (possibly empty) list of TherapyAdministrationMention instances, which contain:
            delivery_status = DeliveryStatus (Choice Value)
                # Delivery status of this dose. ADMINISTERED: actual receipt. PLANNED / HELD / CANCELLED: planned protocol doses, held doses and cancelled orders remain separate from receipt.
                # Possible Values:
                #   - ADMINISTERED
                #   - PLANNED
                #   - HELD
                #   - CANCELLED
                #   - EXPLICITLY_NOT_RECEIVED
                #   - NONE_OF_THE_ABOVE
            phase = TreatmentPhase (Choice Value)
                # Documented phase of this dose; never infer from drug name alone.
                # Possible Values:
                #   - INDUCTION
                #   - CONSOLIDATION
                #   - POST_CONSOLIDATION
                #   - SALVAGE
                #   - NONE_OF_THE_ABOVE
            cycle_name = Optional mention of [Plaintext string]
                # Cycle linked to this administration, if stated.
            administration_date = Optional mention of [Plaintext string]
                # Administration date, ISO YYYY-MM-DD. Null if not stated.
            administration_date_precision = Optional mention of [DatePrecision (Choice Value)]
                # Precision for administration_date. Null when it is null.
                # Possible Values:
                #   - DAY
                #   - MONTH
                #   - YEAR
            dose_amount = Optional mention of [Decimal number 0 or greater]
                # Numeric dose amount as documented. Use the per-body-surface-area amount when the note gives g/m2 or mg/m2; otherwise use the administered amount. Null if no numeric dose is stated.
            dose_unit = Optional mention of [Plaintext string]
                # Dose unit exactly as written, including denominator when present, such as 'g/m2', 'mg/m2', or 'mg/kg'. Null when dose_amount is null.
            route = Optional mention of [Plaintext string]
                # Administration route as written, such as intravenous, intrathecal, or oral. Null if not stated.
            high_dose_methotrexate_explicit_bool = Optional mention of [True or False]
                # True only when this administration is explicitly called high-dose methotrexate. False only when explicitly characterized as not high-dose; null when unknown. Numeric dose-based classification occurs downstream. This label does not establish receipt; inspect delivery_status.
